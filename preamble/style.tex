% Shared pgfplots look: light, no box, Tufte-ish

  book plot/.style={

    tick align=outside,

    /pgf/number format/1000 sep={},
    xlabel style={font=\small},
    ylabel style={font=\small},
    tick label style={font=\footnotesize},
    clip=false,
  }
}
